4. CSS Examples

/* Use data attributes for better isolation */ [data-gamepad-focused="true"] { box-shadow: 0 0 0 3px #007bff; transform: scale(1.02); } [data-gamepad-selected="true"] { background: #007bff; color: white; } /* Context-specific styling */ [data-gamepad-context="grid"] [data-gamepad-focused="true"] { background: rgba(0, 123, 255, 0.1); } /* Respect existing focus styles */ .existing-focus[data-gamepad-focused="true"] { outline: 2px solid #28a745; box-shadow: 0 0 0 2px #28a745, 0 0 0 5px rgba(40, 167, 69, 0.3); }
