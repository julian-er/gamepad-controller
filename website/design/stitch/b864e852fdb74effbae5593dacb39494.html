// 1. Import Spatial Controller Engine
import { createGamepadManager } from 'gamepad-controller';

// 2. Initialize manager with hardware deadzone
const manager = createGamepadManager({
  deadzone: 0.18,            // Radial filter eliminates stick drift
  focusClass: 'ring-focus',    // Visual ring applied to active element
  autoScroll: true,           // 60 FPS smooth viewport auto-scroll
  soundFeedback: true,        // Optional tactile click tick audio
  onSelect: (el) => {
    console.log('Selected target:', el.id);
  }
});

// 3. Start high-frequency 60 FPS RAF loop
manager.start();

import { useGamepadNavigation } from 'gamepad-controller/react';

export function LivingRoomMediaHub() {

    trapFocus: false,
    onButtonDown: (button) => {
      if (button === 'A') launchStream();
      if (button === 'B') history.back();
    }
  });

  return (
    <div className="grid grid-cols-4 gap-6 p-10">
      {shows.map((item) => (
        <button key={item.id} data-spatial className="show-tile">

        </button>
      ))}
    </div>
  );
}
